cmake_minimum_required(VERSION 3.20)
project(DichroicRaw LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# LibRaw ships alternative _ph implementations alongside the normal sources.
# Follow its pinned upstream build manifest, rather than glob duplicate symbols.
file(READ "${RAW_SOURCE}/Makefile.am" RAW_BUILD)
string(REGEX MATCH "lib_libraw_a_SOURCES =([^\r\n]|[\r\n][ \t]+)*" RAW_SECTION "${RAW_BUILD}")
string(REGEX MATCHALL "src/[a-zA-Z0-9_./]+\\.cpp" RAW_SOURCES "${RAW_SECTION}")
list(LENGTH RAW_SOURCES RAW_COUNT)
if(RAW_COUNT LESS 70)
  message(FATAL_ERROR "Could not read the pinned LibRaw source manifest")
endif()
list(TRANSFORM RAW_SOURCES PREPEND "${RAW_SOURCE}/")
file(GLOB LCMS_SOURCES CONFIGURE_DEPENDS "${LCMS_SOURCE}/src/*.c")
add_library(dichroic_lcms STATIC ${LCMS_SOURCES})
target_include_directories(dichroic_lcms PUBLIC "${LCMS_SOURCE}/include")
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(WITH_SIMD OFF CACHE BOOL "" FORCE)
set(WITH_TURBOJPEG OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${JPEG_SOURCE}" jpeg EXCLUDE_FROM_ALL)
add_library(raw_native STATIC raw_native.cpp ${RAW_SOURCES})
target_include_directories(raw_native PUBLIC include "${RAW_SOURCE}" PRIVATE "${JPEG_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/jpeg")
target_compile_definitions(raw_native PRIVATE LIBRAW_NODLL LIBRAW_NOTHREADS USE_LCMS2 USE_JPEG USE_JPEG8)
target_compile_options(raw_native PRIVATE -O3 -ffp-contract=off)
target_link_libraries(raw_native PRIVATE dichroic_lcms jpeg-static)
if(WIN32)
  target_link_libraries(raw_native PRIVATE ws2_32)
endif()
option(BUILD_RAW_PROBE "Build the desktop RAW parity probe" OFF)
if(BUILD_RAW_PROBE)
  add_executable(raw_probe raw_probe.cpp)
  target_link_libraries(raw_probe PRIVATE raw_native)
endif()
